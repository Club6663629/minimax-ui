#!/bin/bash
# 246 生成节点 ComfyUI 启动脚本（CUDA_VISIBLE_DEVICES=0 实际=4090，sage attention）
kill $(pgrep -f 'main.py --listen 0.0.0.0 --port 8188') 2>/dev/null
sleep 2
cd /data/ComfyUI && CUDA_VISIBLE_DEVICES=0 nohup ./venv/bin/python main.py --listen 0.0.0.0 --port 8188 --use-sage-attention --vram-headroom 2 > /tmp/comfyui_8188.log 2>&1 &
echo started pid=$!
